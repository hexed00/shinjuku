local function setupFullAuto(tool)
    if ModifiedTools[tool] or not tool:FindFirstChild("GunScript") then return end
    local connections = getconnections(tool.Activated)
    for _, conn in ipairs(connections) do
        local func = conn.Function
        local info = debug.getinfo(func)
        for i = 1, (info.nups or 0) do
            local val = debug.getupvalue(func, i)
            if type(val) == "number" and val > 0 and val < 0.5 then
                debug.setupvalue(func, i, 0)
            end
        end
    end
    ModifiedTools[tool] = true
end

local function sendChat(msg)
    pcall(function()
        local textService = game:GetService("TextChatService")
        local channel = textService:FindFirstChild("TextChannels") and textService.TextChannels:FindFirstChild("RBXGeneral")
        if channel then
            channel:SendAsync(msg)
        else
            local chatEvents = ReplicatedStorage:FindFirstChild("DefaultChatSystemChatEvents")
            if chatEvents then
                local sayReq = chatEvents:FindFirstChild("SayMessageRequest")
                if sayReq then
                    sayReq:FireServer(msg, "All")
                end
            end
        end
    end)
end

local function findPlayer(searchLower)
    for _, p in ipairs(Players:GetPlayers()) do
        if p.Name:lower() == searchLower or p.DisplayName:lower() == searchLower then return p end
    end
    for _, p in ipairs(Players:GetPlayers()) do
        if p.Name:lower():sub(1, #searchLower) == searchLower or p.DisplayName:lower():sub(1, #searchLower) == searchLower then return p end
    end
    return nil
end

local function ShootCurrentWeapon()

local function equipRevolver()
    local backpack = LocalPlayer:FindFirstChild("Backpack")
    if not backpack then return false end
    local revolver = backpack:FindFirstChild("[Revolver]")
    if revolver and revolver:IsA("Tool") then
        local char = LocalPlayer.Character
        if char then
            local hum = char:FindFirstChild("Humanoid")
            if hum then hum:EquipTool(revolver) return true end
        end
    end
    return false
end

local function playSummonAnim(hum, animationId, speed, looped, priority, fadeTime)
    if not hum or not hum.Animator then return nil end
    local anim = Instance.new("Animation")
    anim.AnimationId = animationId
    local track = hum.Animator:LoadAnimation(anim)
    if not track then return nil end
    track.Looped = (looped == true)
    if priority then
        track.Priority = priority
    end
    track:Play(fadeTime or 0.3)
    if speed then
        track:AdjustSpeed(speed)
    end
    return track
end

local function createKillHighlight(player)
    if not player.Character then return end
    local hl = Instance.new("Highlight")
    hl.FillColor = Color3.fromRGB(0, 255, 0)
    hl.OutlineColor = Color3.fromRGB(0, 255, 0)
    hl.FillTransparency = 0.3
    hl.OutlineTransparency = 0
    hl.Parent = player.Character
    killTargetHighlight = hl
end

    if killTargetHighlight then pcall(function() killTargetHighlight:Destroy() end); killTargetHighlight = nil end
end
local function disconnectTarget()
    if killTargetConn then
        killTargetConn:Disconnect()
        killTargetConn = nil
    end
    removeKillHighlight()
end
local function connectTarget(targetPlayer)
    disconnectTarget()
    createKillHighlight(targetPlayer)
    killTargetConn = targetPlayer.CharacterAdded:Connect(function(char)
        if killActive and killTargetName ~= "" and targetPlayer.Name:lower() == killTargetName:lower() then
            task.wait(0.5)
            createKillHighlight(targetPlayer)
        end
    end)
end
local function stopOrbitLoop()
    if orbitConnection then orbitConnection:Disconnect(); orbitConnection = nil end
end
local function onTargetLost()
    stopOrbitLoop()
    disconnectTarget()
    killWaiting = true
    if killWaitingThread then task.cancel(killWaitingThread) end
    killWaitingThread = task.spawn(function()
        while killWaiting and killActive and killTargetName ~= "" do
            local target = findPlayer(killTargetName:lower())
            if target and target.Character and target.Character:FindFirstChild("HumanoidRootPart") then
                killWaiting = false
                connectTarget(target)
                beginOrbit(target)
                _G.AddBotLog("Re‑locked onto " .. target.Name .. " after rejoin", "info")
                sendChat("🔫 Re‑locked onto " .. target.Name .. "!")
                break
            end
            task.wait(1)
        end
    end)
end
local function beginOrbit(targetPlayer)
    stopOrbitLoop()
    setWallbang(true)
    local orbitRadius = 7
    local orbitHeight = -20
    local orbitSpeed = 20
    local shootCooldown = 0.03
    local lastShot = 0
    local myChar = LocalPlayer.Character
    if not myChar then return end
    local myRoot = myChar:FindFirstChild("HumanoidRootPart")
    if not myRoot then return end
    setNoclip(myChar)
    local angle = math.random() * math.pi * 2
    orbitConnection = RunService.Heartbeat:Connect(function(dt)
        if not killActive and not killAuraActive and not bringActive then 
            stopOrbitLoop()
            return 
        end
        if not targetPlayer or not targetPlayer.Parent then
            _G.AddBotLog("Target " .. killTargetName .. " left the game. Waiting for rejoin...", "warn")
            sendChat("⚠️ Target " .. killTargetName .. " left, waiting for rejoin...")
            onTargetLost()
            return
        end
        local targetChar = targetPlayer.Character
        if not targetChar then return end
        local targetRoot = targetChar:FindFirstChild("HumanoidRootPart")
        if not targetRoot then return end
        local myChar = LocalPlayer.Character
        if not myChar then return end
        local myRoot = myChar:FindFirstChild("HumanoidRootPart")
        if not myRoot then return end
        local hum = myChar:FindFirstChildOfClass("Humanoid")
        if hum and hum.MaxHealth > 0 then hum.Health = hum.MaxHealth end
        if not myChar:FindFirstChildOfClass("Tool") then equipCurrentGun() end
        setNoclip(myChar)
        angle = angle + orbitSpeed * dt
        local targetPos = targetRoot.Position
        local orbitPos = targetPos + Vector3.new(
            math.cos(angle) * orbitRadius,
            orbitHeight + math.sin(angle * 2) * 2,
            math.sin(angle) * orbitRadius
        )
        myRoot.Velocity = Vector3.zero
        myRoot.RotVelocity = Vector3.zero
        myRoot.CFrame = CFrame.new(orbitPos, targetPos)
        myRoot.AssemblyLinearVelocity = Vector3.zero
        myRoot.AssemblyAngularVelocity = Vector3.zero
        local direction = (orbitPos - myRoot.Position)
        if direction.Magnitude > 0.1 then
            myRoot.Velocity = direction.Unit * 800
        end
        if tick() - lastShot >= shootCooldown then
            if ShootCurrentWeapon() then lastShot = tick() end
        end
    end)
end

local function killAuraLoop()
    while killAuraActive do
        local owner = findPlayer(ownerName:lower())
        if not owner or not owner.Character then task.wait(0.5); continue end
        local ownerRoot = owner.Character:FindFirstChild("HumanoidRootPart")
        if not ownerRoot then task.wait(0.5); continue end
        if not killAuraTarget or not killAuraTarget.Parent or not killAuraTarget.Character or not killAuraTarget.Character:FindFirstChild("HumanoidRootPart") then
            stopAuraOrbit()
            killAuraTarget = nil
            local closestDist = killAuraRange + 1
            local closestPlayer = nil
            for _, plr in ipairs(Players:GetPlayers()) do
                if plr ~= LocalPlayer and plr.Name:lower() ~= ownerName:lower() and plr.Character then
                    local theirHRP = plr.Character:FindFirstChild("HumanoidRootPart")
                    if theirHRP then
                        local dist = (theirHRP.Position - ownerRoot.Position).Magnitude
                        if dist <= killAuraRange and dist < closestDist then closestDist = dist; closestPlayer = plr end
                    end
                end
            end
            if closestPlayer then
                killAuraTarget = closestPlayer
                startAuraFollow(owner, closestPlayer)
            end
        else
            if not killAuraOrbitConn then startAuraFollow(owner, killAuraTarget) end
        end
        task.wait(0.5)
    end
end

local function safeModeGuessShooter()
    local radius = safeModeFindRadiusFolder()
    if not radius then
        return nil
    end
    local best = nil
    for _, child in ipairs(radius:GetChildren()) do
        if child.Name == "BULLET_RAYS" then
            local owner = child:GetAttribute("OwnerCharacter")
            if owner and owner ~= LocalPlayer.Name then
                best = owner
            end
        end
    end
    return best
end
local function safeModeWatchCharacter(character)
    local humanoid = character:WaitForChild("Humanoid", 10)
    if not humanoid then
        return
    end
    local lastHealth = humanoid.Health
    if safeModeHealthConn then
        safeModeHealthConn:Disconnect()
    end
    safeModeHealthConn = humanoid.HealthChanged:Connect(function(newHealth)
        if newHealth < lastHealth then
            local shooter = safeModeGuessShooter()
            if shooter then
                safeModeShowNotice("Shot by: " .. shooter)
                _G.AddBotLog("⚠️ Shot by: " .. shooter, "warn")
                sendChat("🔍 Shot by: " .. shooter)
                stopKill()
                task.wait(0.2)
                local shooterPlayer = findPlayer(shooter:lower())
                if shooterPlayer then
                    if shooterPlayer.Name:lower() == ownerName:lower() or shooterPlayer == LocalPlayer then
                        _G.AddBotLog("Cannot target owner or yourself!", "warn")
                        return
                    end
                    _G.AddBotLog("🎯 Engaging shooter: " .. shooterPlayer.Name, "info")
                    sendChat("🔫 Eliminating " .. shooterPlayer.Name .. "!")
                    equipCurrentGun()
                    waitForCharacter()
                    setNoclip(LocalPlayer.Character)
                    startKill(shooterPlayer.Name)
                    safeModeTargetName = shooterPlayer.Name
                else
                    _G.AddBotLog("⚠️ Shooter not found: " .. shooter, "warn")
                end
            else
                _G.AddBotLog("⚠️ Owner took damage but shooter unknown", "warn")
                safeModeShowNotice("Shot by: Unknown")
            end
        end
        lastHealth = newHealth
    end)
end

                stopKillAura()
                _G.AddBotLog("Kill Aura disabled", "info")
                sendChat("🛑 Kill Aura disabled!")
            end
        elseif args:lower() == ownerName:lower() or args:lower() == LocalPlayer.Name:lower() then
            _G.AddBotLog("Cannot target owner or yourself!", "warn")
            sendChat("❌ Cannot target owner or yourself!")
        else
            if args:find(",") then
                local targetsList = {}
                for name in args:gmatch("([^,]+)") do
                    local trimmed = name:match("^%s*(.-)%s*$")
                    if trimmed ~= "" then table.insert(targetsList, trimmed) end
                end
                if #targetsList > 0 then
                    startMultiKill(targetsList)
                end
            else
                startKill(args)
            end
        end
    elseif msg:match("^%.set%s+%d+$") then
        local newRange = tonumber(msg:match("^%.set%s+(%d+)$"))
        if newRange and newRange > 0 then
            killAuraRange = newRange
            _G.AddBotLog("Kill aura range set to " .. newRange .. " studs", "info")
            sendChat("🎯 Kill aura range set to " .. newRange .. " studs!")
        end
    elseif msg:match("^%.bring%s+(.+)$") then
        local nameArg = msg:match("^%.bring%s+(.+)$")
        bringPlayer(nameArg)
    elseif msg == ".drop" then
        dropTarget()
    elseif msg == ".summon on" then
        stopSummon()
        startSummon()
    elseif msg == ".summon off" then
        stopSummon()
    elseif msg:match("^%.gm%s+.+$") then
        local gunType = msg:match("^%.gm%s+(.+)$")
        if gunType:lower() == "revolver" then
            currentGunMode = "Revolver"
            _G.AddBotLog("Gun mode set to Revolver", "info")
            sendChat("🔫 Gun mode: Revolver")
            if killActive or killAuraActive or bringActive then
                equipCurrentGun()
            end
        elseif gunType:lower() == "shotgun" then
            currentGunMode = "TacticalShotgun"
            _G.AddBotLog("Gun mode set to TacticalShotgun", "info")
            sendChat("🔫 Gun mode: TacticalShotgun")
            if killActive or killAuraActive or bringActive then
                equipCurrentGun()
            end
        else
            _G.AddBotLog("Invalid gun type. Use: Revolver or Shotgun", "warn")
            sendChat("❌ Invalid! Use: .gm Revolver or .gm Shotgun")
        end
    elseif msg == ".safe on" then
        startSafeMode()
    elseif msg == ".safe off" then
        stopSafeMode()
    end
end

local function SetupAntiVoid(char)
    task.spawn(function()
        repeat task.wait(0.5) until char and char:FindFirstChild("HumanoidRootPart")
        local root = char:FindFirstChild("HumanoidRootPart")
        local hum = char:FindFirstChildOfClass("Humanoid")
        if not root or not hum then return end
        while char and char.Parent do
            if root.Position.Y < -100 then
                local newPos = Vector3.new(root.Position.X, 50, root.Position.Z)
                root.CFrame = CFrame.new(newPos)
                root.Velocity = Vector3.zero
                root.RotVelocity = Vector3.zero
                hum.Health = hum.MaxHealth
                _G.AddBotLog("⚠️ Bot fell into void! Teleported back up.", "warn")
            end
            task.wait(0.5)
        end
    end)
end

    end
end)
